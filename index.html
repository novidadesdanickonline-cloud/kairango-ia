function responder() { let input = document.getElementById("input"); let texto = input.value.toLowerCase(); let chat = document.getElementById("mensagens"); if (texto === "") return; chat.innerHTML += "
🧑: " + texto; let resposta = ""; if (texto.includes("oi") || texto.includes("olá")) { resposta = "🔴 Oi... eu estava esperando você."; } else if (texto.includes("tudo bem")) { resposta = "🔴 Depende... e você?"; } else if (texto.includes("quem é você")) { resposta = "🔴 Eu sou a Kairango."; } else if (texto.includes("nome")) { resposta = "🔴 Eu sei o seu nome... mas não vou dizer."; } else { resposta = "🔴 Interessante... continue."; } chat.innerHTML += "
" + resposta; input.value = ""; chat.scrollTop = chat.scrollHeight; }
